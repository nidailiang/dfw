{选股公式参数设置}
{N1:=7;    {前一天涨幅百分比}
{N2:=20;   {前N2个交易日}
{N3:=20;    {前N3个交易日}
{N4:=-1;   {当天涨幅下限，可为负数}
{N5:=4;    {当天涨幅上限}

{基础指标计算}
RSV:=(CLOSE-LLV(LOW,9))/(HHV(HIGH,9)-LLV(LOW,9))*100;
K:=SMA(RSV,3,1);
D:=SMA(K,3,1);
J:=3*K-2*D;
STL:=EMA(EMA(C,10),10); {短期趋势线}
MULTILINE:=(MA(CLOSE,14)+MA(CLOSE,28)+MA(CLOSE,57)+MA(CLOSE,114))/4; {多空线}
CHANGEPCT:=(CLOSE-REF(CLOSE,1))/REF(CLOSE,1)*100; {当日涨幅}
AMPLITUDE:=(HIGH-LOW)/REF(CLOSE,1)*100; {当日振幅}

{前一天条件1：收盘价比前N2个交易日的最高价高}
HIGH_N2:=REF(HHV(HIGH,N2),2);
COND8_YEST:=REF(CLOSE,1) > HIGH_N2;

{前一天条件2：成交量比前N3个交易日的成交量高}
VOL_N3:=REF(HHV(VOL,N3),2);
COND9_YEST:=REF(VOL,1) > VOL_N3;

{前一天条件3：涨幅超过N1%}
COND2_YEST:=REF(CHANGEPCT,1) > N1;

{=========== 定义需要排除的板块列表 ===========}
ISBLOCK1:=STRCMP(HYBLOCK,'物流') OR
STRCMP(HYBLOCK,'房地产开发') OR
STRCMP(HYBLOCK,'服装家纺') OR
STRCMP(HYBLOCK,'教育培训') OR
STRCMP(HYBLOCK,'房产服务');

ISBLOCK2:=STRCMP(HYBLOCK,'环境监测') OR
STRCMP(HYBLOCK,'公路铁路') OR
STRCMP(HYBLOCK,'调味品') OR
STRCMP(HYBLOCK,'环境治理') OR
STRCMP(HYBLOCK,'装修装饰');

ISBLOCK3:=STRCMP(HYBLOCK,'工程咨询服务') OR
STRCMP(HYBLOCK,'专业工程') OR
STRCMP(HYBLOCK,'基础建设') OR
STRCMP(HYBLOCK,'轨交设备') OR
STRCMP(HYBLOCK,'医疗美容') OR
STRCMP(HYBLOCK,'摩托车及其他') OR
STRCMP(HYBLOCK,'专业连锁') OR
STRCMP(HYBLOCK,'商业物业经营') OR
STRCMP(HYBLOCK,'文娱用品') OR
STRCMP(HYBLOCK,'家居用品');

ISBLOCK4:=STRCMP(HYBLOCK,'动物保健') OR
STRCMP(HYBLOCK,'农产品加工') OR
STRCMP(HYBLOCK,'装饰建材') OR
STRCMP(HYBLOCK,'油服工程') OR
STRCMP(HYBLOCK,'油气开采');

ISBLOCK5:=STRCMP(HYBLOCK,'环保设备') OR
STRCMP(HYBLOCK,'酒店餐饮') OR
STRCMP(HYBLOCK,'房屋建设') OR
STRCMP(HYBLOCK,'饰品') OR
STRCMP(HYBLOCK,'包装印刷') OR
STRCMP(HYBLOCK,'造纸');

ISBLOCK6:=STRCMP(HYBLOCK,'军工电子');

{合并所有排除板块}
ISBLOCK:=ISBLOCK1 OR ISBLOCK2 OR ISBLOCK3 OR ISBLOCK4 OR ISBLOCK5 OR ISBLOCK6;

{当天条件1：涨幅在N4到N5之间}
COND_TODAY1:=CHANGEPCT >= N4 AND CHANGEPCT <= N5;

{当天条件2：成交量不超过前一天}
COND_TODAY2:=VOL <= REF(VOL,1)*0.6;

{选股条件}
SELECT:NOT(ISBLOCK) AND COND2_YEST AND COND8_YEST AND COND9_YEST AND COND_TODAY1 AND COND_TODAY2;
