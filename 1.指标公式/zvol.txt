{=========== 量价关系公式 ===========}

HH:=1; {1表示高位放量,默认1}

{基础变量定义}
VOLUME:VOL,VOLSTICK;
MAVOL1:=MA(VOLUME,5);
MAVOL2:=MA(VOLUME,10);

{计算涨跌幅}
涨跌幅:=(CLOSE-REF(CLOSE,1))/REF(CLOSE,1)*100;

{计算振幅}
振幅:=(HIGH-LOW)/REF(CLOSE,1)*100;

{根据股票前缀判断是否为创科北}
创业板:=IF(CODELIKE('300'), 1, 0) OR IF(CODELIKE('301'), 1, 0); {或者使用：STRLEFT(CODE,3)='300'}
科创板:=IF(CODELIKE('688'), 1, 0); {或者使用：STRLEFT(CODE,3)='688'}
北交所:=IF(CODELIKE('920'), 1, 0); {或者使用：STRLEFT(CODE,3)='300'}
是否创科北:=科创板 OR 创业板 OR 北交所;

{===========上涨趋势===========}
短期趋势线:=EMA(EMA(C,10),10);
多空线:=(MA(CLOSE,14)+MA(CLOSE,28)+MA(CLOSE,57)+MA(CLOSE,114))/4;
上涨趋势:=短期趋势线>多空线;



{===========涨跌停公式===========}
涨停价:=REF(CLOSE,1)*(1+IF(是否创科北, 0.2, 0.1)); {考虑创科20%限制}
跌停价:=REF(CLOSE,1)*(1-IF(是否创科北, 0.2, 0.1));
涨停条件:=HIGH>=涨停价 AND CLOSE>=涨停价*0.995; {接近涨停}
跌停条件:=LOW<=跌停价 AND CLOSE<=跌停价*1.005; {接近跌停}
{一字板判断}
一字板涨停:=CLOSE>=涨停价*0.995 AND CLOSE<=涨停价*1.005 AND HIGH-LOW<=涨停价*0.01; {振幅极小}
一字板跌停:=CLOSE<=跌停价*1.005 AND CLOSE>=跌停价*0.995 AND HIGH-LOW<=跌停价*0.01; {振幅极小}


{=========== 倍量条件 ===========}
倍量条件1:=NOT(REF(一字板跌停,1)) AND VOL>REF(VOL,1)*1.8 AND 涨跌幅>2;
倍量条件:=倍量条件1;


{=========== 缩量条件 ===========}
前一天阳线:=REF(CLOSE,1)>REF(OPEN,1);
前3天倍量:=EXIST(VOL>REF(VOL,1)*2,3);
缩量条件:=VOL<REF(VOL,1)*0.5 AND (CLOSE<REF(CLOSE,1)) AND 前一天阳线 AND 前3天倍量 AND 上涨趋势;



{=========== 放量阴线 ===========}
前3天最大阳量:=REF(HHV(IF(CLOSE>OPEN,VOL,0),3),1);
前3天最大阴量:=REF(HHV(IF(CLOSE<OPEN,VOL,0),3),1);
前15天最大阳量:=REF(HHV(IF(CLOSE>OPEN,VOL,0),15),1);
前三天有阳线:=EXIST(REF(CLOSE>OPEN,1),3) OR EXIST(REF(涨停条件,1), 3);
吞没形态:=CLOSE<REF(OPEN,1);

{创建创科北系数}
创科北系数:=IF(是否创科北, 1, 0.7); {创科=1, 非创科=0.7}

动态跌幅阈值:=-3*创科北系数; {创科=-3, 非创科=-2.1}

{出1：强烈出货信号}
出1:前三天有阳线 AND (CLOSE<OPEN) AND 涨跌幅<动态跌幅阈值 AND VOL>前3天最大阳量*1.5 AND VOL>前15天最大阳量*1.1, COLORGREEN,NODRAW;

{出2：相对出货信号}
出2:前三天有阳线 AND (CLOSE<OPEN) AND 涨跌幅<动态跌幅阈值 AND VOL<=前3天最大阳量*1.5 AND VOL>前15天最大阳量*1.2, COLORGREEN,NODRAW;

{出3：实心大绿柱子（实体很长的大阴线）}
{计算影线长度}
上影线长度:=(HIGH-MAX(OPEN,CLOSE))/REF(CLOSE,1)*100;
下影线长度:=(MIN(OPEN,CLOSE)-LOW)/REF(CLOSE,1)*100;

{使用创科北系数调整影线条件}
影线短阈值:=3*创科北系数; {创科=3, 非创科=2.1}
影线短条件:=上影线长度<影线短阈值 AND 下影线长度<影线短阈值;
{使用创科北系数调整振幅阈值}
振幅阈值:=10*创科北系数; {创科=10, 非创科=7}
出3阈值:=1.1*创科北系数; {创科=1.1, 非创科=0.77}
涨跌幅阈值:=-4*创科北系数; {创科=-4, 非创科=-2.8}
出3:(CLOSE<OPEN) AND VOL>前15天最大阳量*1.2 AND 下影线长度<影线短阈值 AND 振幅>振幅阈值 AND 涨跌幅<涨跌幅阈值, COLORGREEN,NODRAW;

{出4：跳空高开}
{使用创科北系数调整最高涨幅条件}
最高涨幅阈值:=7*创科北系数; {创科=7, 非创科=4.9}
跳空幅度阈值:=5*创科北系数; {创科=5, 非创科=3.5}
跳空幅度:=(OPEN/REF(CLOSE,1)-1)*100;
最高涨幅:=(HIGH-REF(CLOSE,1))/REF(CLOSE,1)*100;
出4:(CLOSE<OPEN) AND VOL>前15天最大阳量*1.2 AND (跳空幅度>跳空幅度阈值 OR 最高涨幅>最高涨幅阈值) AND 振幅>振幅阈值 AND 涨跌幅<涨跌幅阈值, COLORGREEN,NODRAW;

{计算相对位置}
相对位置:=(CLOSE/多空线-1)*100;

{出货形态}
基础出货形态:=出1 OR 出2;
高位出货形态:=基础出货形态 AND 相对位置>20;
{具体出货形态}
放量阴线:=IF(HH=1, 高位出货形态, 基础出货形态);
前三天存在出货:=EXIST(REF(放量阴线,1), 3);

延续出:=EXIST(REF(放量阴线,1), 3) AND (CLOSE < OPEN) AND (涨跌幅<-5 OR VOL > REF(前3天最大阴量, 1)*0.8), COLORGREEN;

{一字直接认为是出货}
低开幅度:=(OPEN/REF(CLOSE,1)-1)*100;
最高价跌幅:=(HIGH/REF(CLOSE,1)-1)*100;
低开跌停:=CLOSE<=跌停价*1.005 AND CLOSE>=跌停价*0.995 AND 低开幅度<-7 AND 最高价跌幅<-6;
一字出:=一字板跌停 OR 低开跌停, COLORGREEN,NODRAW;

出货条件:=放量阴线 OR 延续出;


{=========== 地量条件 ===========}
前10天最低量:=REF(LLV(VOL,10),1);

{判断K线形态}
{十字星定义：实体小于0.5%，但整体振幅大于1% 暂时未用到}
是十字星:=ABS(CLOSE-OPEN)/OPEN<0.005 AND (HIGH-LOW)/REF(CLOSE,1)*100>1;

{计算涨跌幅绝对值}
涨跌幅绝对值:=ABS(涨跌幅);

{固定地量条件}
固定地量条件:=VOL<=前10天最低量*1.05;

{动态趋势强度}
趋势强度:(短期趋势线-多空线)/多空线*100,COLORYELLOW,NODRAW;
{动态基础宽松度}
基础宽松度:=IF((CLOSE<OPEN), 1.05, IF((CLOSE>OPEN), 1.15, 1.10));
{动态趋势强度调整}
趋势调整:=IF(趋势强度>3, 0.10, IF(趋势强度>1, 0.05, 0));
{动态涨跌幅调整}
涨跌幅调整:=IF(涨跌幅绝对值<1, 0.05, 0);
{动态最终宽松度}
最终宽松度:基础宽松度+趋势调整+涨跌幅调整,COLORYELLOW,NODRAW;
{计算动态地量条件}
动态地量条件:=VOL<=前10天最低量*最终宽松度;

{考虑成交量与均量的关系}
低于均量条件:=VOL<MA(VOL,5) AND VOL<MA(VOL,10);

前28天未出货:=EXIST(REF(放量阴线,1),28);
前10天有倍量:=EXIST(REF(倍量条件,1),10);

{地量条件}
固定地量:=上涨趋势 AND 固定地量条件 AND 低于均量条件 AND NOT(前28天未出货) AND 前10天有倍量;
动态地量:=上涨趋势 AND 动态地量条件 AND 低于均量条件 AND NOT(前28天未出货) AND 前10天有倍量;




{=========== KDJ指标 ===========}
{计算KDJ指标}
RSV:=(C-LLV(L,9))/(HHV(H,9)-LLV(L,9))*100;
K:=SMA(RSV,3,1);
D:=SMA(K,3,1);
J:=3*K-2*D;

{参数设置}
超卖阈值:=16;    {直接买点阈值}
低位阈值:=40;    {条件买点阈值}
观察周期:=5;     {观察极度超卖的周期}

{买点判断}
J_直接买点:=J<=超卖阈值;
J_条件买点:=J>超卖阈值 AND LLV(J,观察周期)<=0 AND J<=低位阈值;

{最终买点}
固定J买点:=J<=低位阈值;
动态J买点:=J_直接买点 OR J_条件买点;

{涨跌振阈值}
固定涨跌振阈值:=IF(是否创科北, 涨跌幅>=-3 AND 涨跌幅<=3 AND 振幅<=6,涨跌幅>=-1.8 AND 涨跌幅<=1.8 AND 振幅<=3.6);
动态涨跌振阈值:=IF(是否创科北, 涨跌幅>=-4 AND 涨跌幅<=3 AND 振幅<=7,涨跌幅>=-2.05 AND 涨跌幅<=1.8 AND 振幅<=4);

{地量J买点}
固定地量J买点:=固定地量 AND 固定J买点 AND 固定涨跌振阈值, COLORRED;
动态地量J买点:=动态地量 AND 动态J买点 AND 动态涨跌振阈值, COLORRED;




{=========== 单针下30 ===========}
{参数设置}
N1:=3;
N2:=21;
A:=85;
B:=30;

{计算当天指标}
当天短期:=100*(C-LLV(L,N1))/(HHV(C,N1)-LLV(L,N1));
当天长期:=100*(C-LLV(L,N2))/(HHV(C,N2)-LLV(L,N2));

{计算前一日指标}
前一日短期:=REF(当天短期,1);
前一日长期:=REF(当天长期,1);

{计算前两日指标}
前两日短期:=REF(当天短期,2);
前两日长期:=REF(当天长期,2);

{获取总市值}
总市值:=FINANCE(40)/10000;

{定义单针下30条件}
条件1:=当天长期>A;
条件2:=当天短期<B;
条件3:=(前一日短期=100 AND 前一日长期=100) OR (前两日短期=100 AND 前两日长期=100);
条件4:=总市值>50;
条件5:=短期趋势线>多空线;
条件6:=EXIST(REF(倍量条件,1),20);
条件7:=NOT(EXIST(REF(放量阴线,1),20));

单针买点:条件1 AND 条件2 AND 条件3 AND 条件4 AND 条件5 AND 条件6 AND 条件7,COLORRED,NODRAW;

{最终买点条件}
B1买点:=动态地量J买点;

低:REF(MA(VOL,3),1)*0.8,DOTLINE,LINETHICK1;
高:REF(MA(VOL,3),1)*1.2,DOTLINE,LINETHICK1;


{STICKLINE(倍量条件 AND O < C, 0, VOL, 2, 0) COLORRED;   {在成交量柱上方2%处画红色实心标记}
{STICKLINE(出货条件 AND O > C, 0, VOL, 2, 0) COLORLIGREEN;   {在成交量柱上方2%处画红色实心标记}

DRAWICON(缩量条件,VOL*0.92,5); {缩量下跌，5号图标}
DRAWICON(出货条件,VOL*0.98,8); {放量阴线，8号图标}
DRAWICON(B1买点,VOL*0.92,7); {B1，7号图标}
