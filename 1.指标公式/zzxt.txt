VAR1A:=(HHV(HIGH,4)-CLOSE)/(HHV(HIGH,4)-LLV(LOW,4))*100-90;
VAR2A:=SMA(VAR1A,4,1)+100;
VAR3A:=(CLOSE-LLV(LOW,4))/(HHV(HIGH,4)-LLV(LOW,4))*100;
VAR4A:=SMA(VAR3A,6,1);
VAR5A:=SMA(VAR4A,6,1)+100;
VAR6A:=VAR5A-VAR2A;

砖型图:=IF(VAR6A>4,VAR6A-4,0);

{=== 状态定义 ===}
AA:=(REF(砖型图,1)<砖型图);   {今天本该画红柱，上涨}
BB:=REF(砖型图,1)>砖型图;   {今天本该画绿柱，下跌}
CC:=REF(AA,1)=0 AND (AA=1); {绿/0转红的第一天}

{=== 新增计算柱子长度与比例的逻辑 ===}
今红长:= 砖型图 - REF(砖型图,1);         {今天红柱的绝对高度}
昨绿长:= REF(砖型图,2) - REF(砖型图,1);  {昨天绿柱的绝对高度}

{DD: 是转红第一天 + 昨天确实是绿柱 + 今天红柱高度大于等于昨天绿柱的2/3}
DD:= CC AND REF(BB,1) AND (今红长 >= 昨绿长 * 2/3);

{=== 画柱子(STICKLINE) ===}
{1. 画普通绿柱}
STICKLINE(BB, 砖型图, REF(砖型图,1), 3, 0), COLOR00FF00;

{2. 画特殊洋红柱（满足DD条件）}
STICKLINE(DD, 砖型图, REF(砖型图,1), 3, 0), COLORRED;

{3. 画普通红柱（排除掉DD的情况，防止颜色重叠）}
STICKLINE(AA AND (DD=0), 砖型图, REF(砖型图,1), 3, 0), COLORMAGENTA;

红持绿空:=0,COLORGREEN;

{=== 选股信号 ===}
{如果你想专门选出出现洋红柱（反包强势）的股票，请把下面这句改成 XG:=DD; }
XG:=CC>0;
